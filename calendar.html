Appointment Types -- Reusable meeting definitions shown inside GCM OS and, -- later, on the public Global Concepts Media booking page. -- --------------------------------------------------------- CREATE TABLE IF NOT EXISTS calendar_appointment_types ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, slug TEXT NOT NULL UNIQUE, description TEXT, duration_minutes INTEGER NOT NULL DEFAULT 60 CHECK (duration_minutes > 0), maximum_per_day INTEGER CHECK (maximum_per_day IS NULL OR maximum_per_day > 0), preparation_required INTEGER NOT NULL DEFAULT 0 CHECK (preparation_required IN (0, 1)), allow_google_meet INTEGER NOT NULL DEFAULT 1 CHECK (allow_google_meet IN (0, 1)), allow_phone INTEGER NOT NULL DEFAULT 1 CHECK (allow_phone IN (0, 1)), allow_in_person INTEGER NOT NULL DEFAULT 0 CHECK (allow_in_person IN (0, 1)), is_public INTEGER NOT NULL DEFAULT 0 CHECK (is_public IN (0, 1)), is_active INTEGER NOT NULL DEFAULT 1 CHECK (is_active IN (0, 1)), display_order INTEGER NOT NULL DEFAULT 0, created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP ); CREATE INDEX IF NOT EXISTS idx_calendar_appointment_types_active ON calendar_appointment_types (is_active, display_order, name); CREATE INDEX IF NOT EXISTS idx_calendar_appointment_types_public ON calendar_appointment_types (is_public, is_active, display_order); -- --------------------------------------------------------- -- 2. Availability Rules -- One active agency-wide row is seeded for the first release. -- The schema supports future owner/user-specific rule sets. -- Days are stored as JSON integers: -- Sunday=0 through Saturday=6. -- --------------------------------------------------------- CREATE TABLE IF NOT EXISTS calendar_availability_rules ( id INTEGER PRIMARY KEY AUTOINCREMENT, owner_key TEXT NOT NULL DEFAULT 'andy' UNIQUE, timezone TEXT NOT NULL DEFAULT 'America/New_York', available_days_json TEXT NOT NULL DEFAULT '[1,2,3,4,5]', start_time TEXT NOT NULL DEFAULT '09:00', end_time TEXT NOT NULL DEFAULT '17:00', minimum_advance_notice_hours INTEGER NOT NULL DEFAULT 24 CHECK (minimum_advance_notice_hours >= 0), maximum_booking_window_days INTEGER NOT NULL DEFAULT 90 CHECK (maximum_booking_window_days > 0), buffer_before_minutes INTEGER NOT NULL DEFAULT 15 CHECK (buffer_before_minutes >= 0), buffer_after_minutes INTEGER NOT NULL DEFAULT 15 CHECK (buffer_after_minutes >= 0), travel_buffer_minutes INTEGER NOT NULL DEFAULT 60 CHECK (travel_buffer_minutes >= 0), maximum_meetings_per_day INTEGER NOT NULL DEFAULT 4 CHECK (maximum_meetings_per_day > 0), booking_enabled INTEGER NOT NULL DEFAULT 0 CHECK (booking_enabled IN (0, 1)), created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP ); -- --------------------------------------------------------- -- 3. Calendar Appointments -- Durable source of truth for: -- - appointments created in GCM OS -- - future website bookings -- - future Google Calendar synchronization -- No sample or fake appointment rows are inserted. -- --------------------------------------------------------- CREATE TABLE IF NOT EXISTS calendar_appointments ( id INTEGER PRIMARY KEY AUTOINCREMENT, appointment_type_id INTEGER, client_id INTEGER, prospect_id INTEGER, title TEXT NOT NULL, contact_name TEXT, company_name TEXT, contact_email TEXT, contact_phone TEXT, appointment_date TEXT NOT NULL, start_time TEXT NOT NULL, end_time TEXT, timezone TEXT NOT NULL DEFAULT 'America/New_York', location_type TEXT NOT NULL DEFAULT 'phone' CHECK ( location_type IN ( 'google_meet', 'phone', 'in_person' ) ), location_name TEXT, location_address TEXT, meeting_url TEXT, status TEXT NOT NULL DEFAULT 'scheduled' CHECK ( status IN ( 'requested', 'scheduled', 'confirmation_sent', 'confirmed', 'rescheduled', 'completed', 'cancelled', 'no_show' ) ), source_type TEXT NOT NULL DEFAULT 'internal' CHECK ( source_type IN ( 'internal', 'public_booking', 'google_calendar', 'import' ) ), public_booking_reference TEXT UNIQUE, confirmation_sent_at TEXT, confirmed_at TEXT, cancelled_at TEXT, completed_at TEXT, preparation_required INTEGER NOT NULL DEFAULT 0 CHECK (preparation_required IN (0, 1)), follow_up_required INTEGER NOT NULL DEFAULT 0 CHECK (follow_up_required IN (0, 1)), notes TEXT, google_calendar_event_id TEXT UNIQUE, created_by TEXT, updated_by TEXT, created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (appointment_type_id) REFERENCES calendar_appointment_types (id) ON DELETE SET NULL, FOREIGN KEY (client_id) REFERENCES clients (id) ON DELETE SET NULL ); CREATE INDEX IF NOT EXISTS idx_calendar_appointments_date ON calendar_appointments (appointment_date, start_time); CREATE INDEX IF NOT EXISTS idx_calendar_appointments_status_date ON calendar_appointments (status, appointment_date, start_time); CREATE INDEX IF NOT EXISTS idx_calendar_appointments_client ON calendar_appointments (client_id, appointment_date); CREATE INDEX IF NOT EXISTS idx_calendar_appointments_type ON calendar_appointments (appointment_type_id, appointment_date); CREATE INDEX IF NOT EXISTS idx_calendar_appointments_source ON calendar_appointments (source_type, appointment_date); -- --------------------------------------------------------- -- 4. Calendar Reminders -- Standalone day-to-day reminders created directly in Calendar. -- Operational records from Media, Work, Finance, Investigations, -- and Proof remain in their own tables and will later be exposed -- through a combined calendar-feed endpoint. -- --------------------------------------------------------- CREATE TABLE IF NOT EXISTS calendar_reminders ( id INTEGER PRIMARY KEY AUTOINCREMENT, client_id INTEGER, title TEXT NOT NULL, description TEXT, reminder_date TEXT NOT NULL, reminder_time TEXT, timezone TEXT NOT NULL DEFAULT 'America/New_York', priority TEXT NOT NULL DEFAULT 'normal' CHECK ( priority IN ( 'low', 'normal', 'high', 'urgent' ) ), status TEXT NOT NULL DEFAULT 'open' CHECK ( status IN ( 'open', 'completed', 'dismissed', 'cancelled' ) ), recurrence_rule TEXT, source_type TEXT NOT NULL DEFAULT 'calendar' CHECK ( source_type IN ( 'calendar', 'communication', 'investigation', 'work_item', 'media', 'finance', 'proof', 'system' ) ), source_record_id INTEGER, source_url TEXT, completed_at TEXT, dismissed_at TEXT, created_by TEXT, updated_by TEXT, created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (client_id) REFERENCES clients (id) ON DELETE SET NULL ); CREATE INDEX IF NOT EXISTS idx_calendar_reminders_date ON calendar_reminders (reminder_date, reminder_time); CREATE INDEX IF NOT EXISTS idx_calendar_reminders_status_date ON calendar_reminders (status, reminder_date, reminder_time); CREATE INDEX IF NOT EXISTS idx_calendar_reminders_client ON calendar_reminders (client_id, reminder_date); CREATE INDEX IF NOT EXISTS idx_calendar_reminders_source ON calendar_reminders (source_type, source_record_id); -- --------------------------------------------------------- -- 5.
